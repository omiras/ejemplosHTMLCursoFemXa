Valores booleanos

Una condición siempre responde true o false.

let mayorEdad = true;
let aprobado = false;

console.log(18 >= 18); // true
console.log(5 > 10);   // false

La estructura if

let edad = 20;

if (edad >= 18) {
    console.log("Puedes entrar");
}
Idea: el código solo se ejecuta cuando la condición vale true.

if...else

let edad = 15;

if (edad >= 18) {
    console.log("Mayor de edad");
} else {
    console.log("Menor de edad");
}

if...else if

let nota = 8;

if (nota >= 9) {
    console.log("Sobresaliente");
} else if (nota >= 7) {
    console.log("Notable");
} else if (nota >= 5) {
    console.log("Aprobado");
} else {
    console.log("Suspenso");
}

Operadores lógicos

AND (&&)

let ingresos = 1800;
let contratoIndefinido = true;

if (ingresos >= 1500 && contratoIndefinido) {
    console.log("Hipoteca concedida");
}

OR (||)

let vip = false;
let invitado = true;

if (vip || invitado) {
    console.log("Puede entrar");
}

NOT (!)

let matriculado = false;

if (!matriculado) {
    console.log("Todavía puedes inscribirte");
}

switch

let idioma = "medio";

switch (idioma) {
    case "bajo":
        console.log("Necesitas mejorar");
        break;
    case "medio":
        console.log("Puedes seguir el curso");
        break;
    case "alto":
        console.log("Perfecto");
        break;
    default:
        console.log("Valor desconocido");
}

Casos de uso

  • ¿Me concederán la hipoteca?
  • ¿Puede inscribirse al curso?
  • ¿Es mayor de edad?
  • ¿Trabaja al menos 3 días?
  • ¿Tiene un nivel suficiente de español?

Errores habituales

  • No pongas ; justo después del if (...).
  • No olvides los paréntesis.
  • Distingue = (asignación) de == y === (comparación).
